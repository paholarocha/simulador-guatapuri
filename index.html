Código Mejorado para Wokwi (con LCD 16x2, Plotter, LEDs y Buzzer)

Para que tu circuito en Wokwi tenga pantalla LCD, LEDs de estado, alarma sonora, deslizadores y gráficas en el Serial Plotter (el botón con forma de onda abajo a la derecha en Wokwi), reemplaza tus archivos en Wokwi por los siguientes:

1. Archivo: sketch.ino 
#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Inicializar pantalla LCD I2C (0x27, 16 columnas, 2 filas)
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Credenciales Telegram
const char* ssid = "Wokwi-GUEST";
const char* password = "";
String botToken = "8854897042:AAEj_CURG794CRjskPBTMXNuHzM6p-wgwFc";
String chatId   = "7787186027";
unsigned long ultimaAlerta = 0;

void setup() {
  Serial.begin(115200);
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("NT-IoT Guatapuri");
  lcd.setCursor(0, 1);
  lcd.print("Conectando WiFi");

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(300);
  }
  lcd.clear();
}

void loop() {
  int rawTemp = analogRead(34); // Potenciómetro 1: Temperatura
  int rawPH   = analogRead(35); // Potenciómetro 2: pH
  int rawTurb = analogRead(32); // Potenciómetro 3: Turbidez

  float temperatura = map(rawTemp, 0, 4095, 15, 45);
  float ph          = map(rawPH, 0, 4095, 0, 140) / 10.0;
  float turbidez    = map(rawTurb, 0, 4095, 0, 300);

  // Actualizar Pantalla LCD
  lcd.setCursor(0, 0);
  lcd.print("Temp: " + String(temperatura, 1) + " C   ");
  lcd.setCursor(0, 1);
  lcd.print("pH:   " + String(ph, 1) + "       ");

  // Enviar ondas al Serial Plotter de Wokwi
  Serial.print("Temperatura:");
  Serial.print(temperatura);
  Serial.print(",pH:");
  Serial.print(ph);
  Serial.print(",Turbidez_x01:");
  Serial.println(turbidez * 0.1);

  // Alerta automática a Telegram si supera umbral
  if ((temperatura > 35.0 || ph < 6.0 || ph > 8.5) && (millis() - ultimaAlerta > 30000)) {
    if (WiFi.status() == WL_CONNECTED) {
      HTTPClient http;
      String msg = "🚨 Alerta Rio Guatapuri 🚨%0ATemp: " + String(temperatura, 1) + "C%0ApH: " + String(ph, 1);
      http.begin("https://api.telegram.org/bot" + botToken + "/sendMessage?chat_id=" + chatId + "&text=" + msg);
      http.GET();
      http.end();
      ultimaAlerta = millis();
    }
  }

  delay(800);
}
